        document.addEventListener('DOMContentLoaded', function() {
            // Create floating books
            createFloatingBooks();
            
            // DOM Elements
            const sendCodeBtn = document.getElementById('sendCodeBtn');
            const emailInput = document.getElementById('email');
            const verificationCodeInput = document.getElementById('verificationCode');
            const studentNameInput = document.getElementById('studentName');
            const classTeacherInput = document.getElementById('classTeacher');
            const indexNumberInput = document.getElementById('indexNumber');
            const classSelect = document.getElementById('class');
            const streamSelect = document.getElementById('stream');
            const passwordInput = document.getElementById('password');
            const confirmPasswordInput = document.getElementById('confirmPassword');
            const passwordSection = document.getElementById('passwordSection');
            const togglePassword = document.getElementById('togglePassword');
            const toggleConfirmPassword = document.getElementById('toggleConfirmPassword');
            const agreeTermsCheckbox = document.getElementById('agreeTerms');
            const submitButton = document.getElementById('submitButton');
            const messageDisplay = document.getElementById('messageDisplay');
            const subjectsSection = document.getElementById('subjectsSection');
            const compulsoryList = document.getElementById('compulsoryList');
            const electiveList = document.getElementById('electiveList');
            const subjectCounter = document.getElementById('subjectCounter');
            const electiveTitle = document.getElementById('electiveTitle');
            
            let emailVerified = false;
            let countdownInterval;
            let selectedSubjects = [];
            let currentClassLevel = '';

            // Subject data
            const compulsorySubjects = [
                "Mathematics", "English Language", "Physics", "Chemistry", 
                "Biology", "History", "Geography"
            ];

            const electiveSubjects = [
                "Computer Science", "Agriculture", "Literature in English",
                "French", "German", "Kiswahili", "Luganda", "Fine Art", 
                "Performing Arts", "Physical Education", "Technology and Design",
                "Economics", "Religious Education", "Entrepreneurship"
            ];
            
            // Function to create floating books animation
            function createFloatingBooks() {
                const container = document.getElementById('floatingBooks');
                const bookCount = 15;
                const bookIcons = ['book', 'book-open', 'bookmark', 'graduation-cap', 'scroll', 'file-alt'];
                
                // Check for reduced motion preference
                const reducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
                if (reducedMotion) return;
                
                for (let i = 0; i < bookCount; i++) {
                    const book = document.createElement('i');
                    book.classList.add('book', 'fas', `fa-${bookIcons[Math.floor(Math.random() * bookIcons.length)]}`);
                    
                    // Random position and animation
                    const posX = Math.random() * 100;
                    const delay = Math.random() * 10;
                    const duration = Math.random() * 15 + 10;
                    const size = Math.random() * 1 + 1;
                    
                    book.style.left = `${posX}%`;
                    book.style.animationDuration = `${duration}s`;
                    book.style.animationDelay = `${delay}s`;
                    book.style.fontSize = `${size}rem`;
                    
                    container.appendChild(book);
                }
            }
            
            // Toggle password visibility
            function setupPasswordToggle(toggleElement, passwordField) {
                toggleElement.addEventListener('click', function() {
                    const type = passwordField.getAttribute('type') === 'password' ? 'text' : 'password';
                    passwordField.setAttribute('type', type);
                    toggleElement.classList.toggle('fa-eye');
                    toggleElement.classList.toggle('fa-eye-slash');
                });
            }
            
            // Initialize password toggles
            setupPasswordToggle(togglePassword, passwordInput);
            setupPasswordToggle(toggleConfirmPassword, confirmPasswordInput);
            
            // Class selection change handler
            classSelect.addEventListener('change', function() {
                currentClassLevel = this.value;
                if (currentClassLevel) {
                    subjectsSection.style.display = 'block';
                    renderSubjectSelection();
                } else {
                    subjectsSection.style.display = 'none';
                }
                validateForm();
            });
            
            // Render subject selection based on class level
            function renderSubjectSelection() {
                // Clear previous selections
                compulsoryList.innerHTML = '';
                electiveList.innerHTML = '';
                selectedSubjects = [];
                
                // Add compulsory subjects
                compulsorySubjects.forEach(subject => {
                    selectedSubjects.push(subject);
                    const subjectItem = createSubjectItem(subject, true, true);
                    compulsoryList.appendChild(subjectItem);
                });
                
                // Add elective subjects
                let maxElectives = 0;
                if (['S.1', 'S.2'].includes(currentClassLevel)) {
                    maxElectives = 5;
                    electiveTitle.textContent = 'Select 5 Additional Subjects';
                } else if (['S.3', 'S.4'].includes(currentClassLevel)) {
                    maxElectives = 2;
                    electiveTitle.textContent = 'Select 2 Additional Subjects';
                } else if (['S.5', 'S.6'].includes(currentClassLevel)) {
                    maxElectives = 5;
                    electiveTitle.textContent = 'Select 5 Principal Subjects';
                }
                
                electiveSubjects.forEach(subject => {
                    const subjectItem = createSubjectItem(subject, false, false);
                    electiveList.appendChild(subjectItem);
                });
                
                // Update counter
                updateSubjectCounter(maxElectives);
            }
            
            // Create subject checkbox item
            function createSubjectItem(subjectName, isCompulsory, isChecked) {
                const div = document.createElement('div');
                div.className = `subject-item ${isCompulsory ? 'compulsory' : ''}`;
                
                const checkbox = document.createElement('input');
                checkbox.type = 'checkbox';
                checkbox.className = 'subject-checkbox';
                checkbox.id = `subject-${subjectName.replace(/\s+/g, '-')}`;
                checkbox.value = subjectName;
                checkbox.checked = isChecked;
                checkbox.disabled = isCompulsory;
                
                const label = document.createElement('label');
                label.htmlFor = checkbox.id;
                label.textContent = subjectName;
                
                if (!isCompulsory) {
                    checkbox.addEventListener('change', function() {
                        if (this.checked) {
                            selectedSubjects.push(subjectName);
                        } else {
                            const index = selectedSubjects.indexOf(subjectName);
                            if (index !== -1) {
                                selectedSubjects.splice(index, 1);
                            }
                        }
                        
                        // Update counter
                        let maxElectives = 0;
                        if (['S.1', 'S.2'].includes(currentClassLevel)) maxElectives = 5;
                        else if (['S.3', 'S.4'].includes(currentClassLevel)) maxElectives = 2;
                        else if (['S.5', 'S.6'].includes(currentClassLevel)) maxElectives = 5;
                        
                        updateSubjectCounter(maxElectives);
                        validateForm();
                    });
                }
                
                div.appendChild(checkbox);
                div.appendChild(label);
                return div;
            }
            
            // Update subject counter
            function updateSubjectCounter(maxElectives) {
                const totalSubjects = selectedSubjects.length;
                const compulsoryCount = compulsorySubjects.length;
                const electiveCount = totalSubjects - compulsoryCount;
                
                let counterText = '';
                if (['S.1', 'S.2', 'S.3', 'S.4'].includes(currentClassLevel)) {
                    counterText = `${electiveCount}/${maxElectives} Electives Selected`;
                } else {
                    counterText = `${totalSubjects}/${maxElectives} Subjects Selected`;
                }
                
                subjectCounter.textContent = counterText;
                
                // Change color based on selection status
                if ((['S.1', 'S.2'].includes(currentClassLevel) && electiveCount >= maxElectives) {
                    subjectCounter.style.color = '#2ecc71'; // Green
                } else if ((['S.3', 'S.4'].includes(currentClassLevel) && electiveCount >= maxElectives) {
                    subjectCounter.style.color = '#2ecc71'; // Green
                } else if ((['S.5', 'S.6'].includes(currentClassLevel) && totalSubjects >= maxElectives) {
                    subjectCounter.style.color = '#2ecc71'; // Green
                } else {
                    subjectCounter.style.color = '#f39c12'; // Orange
                }
            }
            
            // Validate email format
            function isValidEmail(email) {
                return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
            }
            
            // Validate password complexity
            function isValidPassword(password) {
                return password.length >= 8 && 
                       /[A-Z]/.test(password) && 
                       /[a-z]/.test(password) && 
                       /\d/.test(password) && 
                       /[!@#$%^&*()_+]/.test(password);
            }
            
            // Validate passwords match
            function doPasswordsMatch() {
                return passwordInput.value === confirmPasswordInput.value;
            }
            
            // Validate form and update button states
            function validateForm() {
                const isEmailValid = isValidEmail(emailInput.value);
                const isNameFilled = studentNameInput.value.trim() !== '';
                const isTeacherFilled = classTeacherInput.value.trim() !== '';
                const isIndexFilled = indexNumberInput.value.trim() !== '';
                const isClassSelected = classSelect.value !== '';
                const isStreamSelected = streamSelect.value !== '';
                const isCodeFilled = verificationCodeInput.value.length === 6;
                const isTermsChecked = agreeTermsCheckbox.checked;
                
                // Enable/disable send code button based on email validity
                sendCodeBtn.disabled = !isEmailValid;
                
                let canSubmit = false;
                
                if (!emailVerified) {
                    // Phase 1: Email verification
                    canSubmit = isNameFilled && isTeacherFilled && isIndexFilled &&
                                isClassSelected && isStreamSelected &&
                                isEmailValid && isCodeFilled && isTermsChecked;
                } else {
                    // Phase 2: Registration
                    const isPasswordValid = isValidPassword(passwordInput.value);
                    const passwordsMatch = doPasswordsMatch();
                    canSubmit = isNameFilled && isTeacherFilled && isIndexFilled &&
                                isClassSelected && isStreamSelected &&
                                isEmailValid && isTermsChecked &&
                                isPasswordValid && passwordsMatch;
                }
                
                // Update submit button state
                submitButton.disabled = !canSubmit;
                if (canSubmit) {
                    submitButton.classList.add('active');
                } else {
                    submitButton.classList.remove('active');
                }
            }
            
            // Show message
            function showMessage(message, type = 'info') {
                messageDisplay.textContent = message;
                messageDisplay.className = `message-display ${type}`;
                messageDisplay.style.display = 'block';
                
                // Auto-hide after 5 seconds
                setTimeout(() => {
                    messageDisplay.style.display = 'none';
                }, 5000);
            }
            
            // Handle email verification
            sendCodeBtn.addEventListener('click', function() {
                if (!isValidEmail(emailInput.value)) {
                    showMessage('Please enter a valid email address.', 'error');
                    return;
                }
                
                // Simulate sending verification code
                showMessage('Verification code sent to your email.', 'success');
                
                // Simulate countdown for resend
                let seconds = 60;
                sendCodeBtn.disabled = true;
                sendCodeBtn.textContent = `Resend in ${seconds}s`;
                
                clearInterval(countdownInterval);
                countdownInterval = setInterval(function() {
                    seconds--;
                    sendCodeBtn.textContent = `Resend in ${seconds}s`;
                    
                    if (seconds <= 0) {
                        clearInterval(countdownInterval);
                        sendCodeBtn.disabled = false;
                        sendCodeBtn.textContent = 'Send Verification Code';
                    }
                }, 1000);
            });
            
            // Handle form submission
            submitButton.addEventListener('click', function() {
                if (!emailVerified) {
                    // Verify code
                    if (verificationCodeInput.value === "123456") { // Static code for demo
                        emailVerified = true;
                        passwordSection.style.display = 'block';
                        showMessage('Email verified successfully!', 'success');
                        submitButton.textContent = 'Create Account';
                        
                        // Scroll to password section
                        passwordSection.scrollIntoView({ behavior: 'smooth', block: 'center' });
                    } else {
                        showMessage('Invalid verification code. Please try again.', 'error');
                    }
                } else {
                    // Final submission
                    if (!isValidPassword(passwordInput.value)) {
                        showMessage('Password must be at least 8 characters with uppercase, lowercase, number, and special character.', 'error');
                        return;
                    }
                    
                    if (!doPasswordsMatch()) {
                        showMessage('Passwords do not match.', 'error');
                        return;
                    }
                    
                    // Show success message
                    showMessage('Account created successfully! Redirecting to login...', 'success');
                    
                    // Simulate redirect
                    setTimeout(() => {
                        window.location.href = 'studentlogin.html';
                    }, 2000);
                }
                
                validateForm();
            });
            
            // Form validation on input
            studentNameInput.addEventListener('input', validateForm);
            classTeacherInput.addEventListener('input', validateForm);
            indexNumberInput.addEventListener('input', validateForm);
            streamSelect.addEventListener('change', validateForm);
            emailInput.addEventListener('input', validateForm);
            verificationCodeInput.addEventListener('input', validateForm);
            passwordInput.addEventListener('input', validateForm);
            confirmPasswordInput.addEventListener('input', validateForm);
            agreeTermsCheckbox.addEventListener('change', validateForm);
            
            // Initial validation
            validateForm();
        });
    </script>
