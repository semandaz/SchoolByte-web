    <script>
        document.addEventListener('DOMContentLoaded', function() {
            // --- DOM ELEMENTS ---
            const form = document.getElementById('studentSignupForm');
            const allInputs = form.querySelectorAll('input, select');
            const sendCodeBtn = document.getElementById('sendCodeBtn');
            const emailInput = document.getElementById('email');
            const verificationCodeInput = document.getElementById('verificationCode');
            const classSelect = document.getElementById('class');
            const passwordSection = document.getElementById('passwordSection');
            const subjectsSection = document.getElementById('subjectsSection');
            const submitButton = document.getElementById('submitButton');
            const messageDisplay = document.getElementById('messageDisplay');
            
            // Error message elements
            const indexNumberError = document.getElementById('indexNumberError');
            const passwordError = document.getElementById('passwordError');
            const confirmPasswordError = document.getElementById('confirmPasswordError');
            const subjectError = document.getElementById('subjectError');

            // State variables
            let emailVerified = false;

            // --- SUBJECT DATA ---
            const compulsorySubjectsList = ["Mathematics", "English Language", "Physics", "Chemistry", "Biology", "History", "Geography"];
            const allSubjectsList = [
                "Mathematics", "English Language", "Biology", "Chemistry", "Physics", 
                "History", "Geography", "Computer Science", "Agriculture", "Literature in English",
                "French", "German", "Kiswahili", "Luganda", "Fine Art", "Performing Arts",
                "Physical Education", "Technology and Design", "Economics", "General Paper", "Subsidiary Mathematics"
            ];
            const electiveSubjectsList = allSubjectsList.filter(s => !compulsorySubjectsList.includes(s));

            // --- INITIALIZATION ---
            createFloatingBooks();
            setupEventListeners();
            validateForm();

            // --- EVENT LISTENERS ---
            function setupEventListeners() {
                // Validate form on any input change
                allInputs.forEach(input => input.addEventListener('input', validateForm));
                classSelect.addEventListener('change', handleClassChange);

                // Button clicks
                sendCodeBtn.addEventListener('click', simulateSendCode);
                submitButton.addEventListener('click', handleFormSubmit);

                // Password visibility toggles
                document.querySelectorAll('.password-toggle').forEach(toggle => {
                    toggle.addEventListener('click', () => {
                        const targetInput = document.getElementById(toggle.dataset.target);
                        const type = targetInput.getAttribute('type') === 'password' ? 'text' : 'password';
                        targetInput.setAttribute('type', type);
                        toggle.classList.toggle('fa-eye');
                        toggle.classList.toggle('fa-eye-slash');
                    });
                });
            }

            // --- CORE LOGIC ---

            /**
             * Main handler for the "Continue" / "Register" button.
             * Switches between verifying the code and submitting the final form.
             */
            function handleFormSubmit() {
                if (!emailVerified) {
                    // --- Stage 1: Verify Email Code ---
                    // This is a simulation, no actual backend call
                    showMessage('Email successfully verified!', 'success');
                    emailVerified = true;

                    // Update UI for the next stage
                    passwordSection.style.display = 'block';
                    subjectsSection.style.display = 'block';
                    handleClassChange(); // Populate subjects based on current class selection

                    // Lock down previous fields
                    emailInput.readOnly = true;
                    verificationCodeInput.readOnly = true;
                    sendCodeBtn.style.display = 'none';
                    
                    submitButton.innerHTML = 'Register Account <i class="fas fa-user-plus"></i>';
                } else {
                    // --- Stage 2: Final Form Submission ---
                    // This is the end of the prototype flow
                    showMessage('Congratulations! Account created.', 'success');
                    submitButton.disabled = true;
                    submitButton.innerHTML = 'Registered! <i class="fas fa-check"></i>';
                }
                validateForm(); // Re-validate to update button state
            }
            
            /**
             * Simulates sending a verification code.
             */
            function simulateSendCode() {
                showMessage('A verification code has been "sent" to your email.', 'info');
                sendCodeBtn.disabled = true;
                sendCodeBtn.textContent = 'Code Sent';
            }

            /**
             * Handles changes to the class dropdown to update the subject selection UI.
             */
            function handleClassChange() {
                if (!emailVerified) return; // Don't show subjects until email is verified
                
                const selectedClass = classSelect.value;
                let rules = { compulsory: [], electives: [], maxElectives: 0 };

                if (['S.1', 'S.2'].includes(selectedClass)) {
                    rules = { compulsory: compulsorySubjectsList, electives: electiveSubjectsList, maxElectives: 5 };
                } else if (['S.3', 'S.4'].includes(selectedClass)) {
                    rules = { compulsory: compulsorySubjectsList, electives: electiveSubjectsList, maxElectives: 2 };
                } else if (['S.5', 'S.6'].includes(selectedClass)) {
                    // A-Level has different rules, for this prototype we'll simplify
                    rules = { compulsory: ["General Paper"], electives: allSubjectsList.filter(s => s !== "General Paper"), maxElectives: 4 };
                }
                
                populateSubjects(rules);
                validateForm();
            }

            /**
             * Renders the subject checkboxes based on class rules.
             * @param {object} rules - The rules for the selected class.
             */
            function populateSubjects({ compulsory, electives, maxElectives }) {
                const compulsoryContainer = document.getElementById('compulsoryList');
                const electiveContainer = document.getElementById('electiveList');
                
                compulsoryContainer.innerHTML = '';
                electiveContainer.innerHTML = '';
                document.getElementById('electiveTitle').textContent = `Additional Subjects (Select ${maxElectives})`;

                compulsory.forEach(subject => {
                    compulsoryContainer.appendChild(createSubjectCheckbox(subject, true, true));
                });

                electives.forEach(subject => {
                    const checkbox = createSubjectCheckbox(subject, false, false);
                    checkbox.addEventListener('change', () => {
                        validateSubjectSelection(maxElectives);
                        validateForm();
                    });
                    electiveContainer.appendChild(checkbox);
                });
                
                validateSubjectSelection(maxElectives); // Initial check
            }

            /**
             * Creates a single subject checkbox item.
             */
            function createSubjectCheckbox(name, isChecked, isDisabled) {
                const id = `subject-${name.replace(/\s+/g, '-')}`;
                const item = document.createElement('div');
                item.className = 'subject-item';
                item.innerHTML = `
                    <input type="checkbox" id="${id}" value="${name}" ${isChecked ? 'checked' : ''} ${isDisabled ? 'disabled' : ''}>
                    <label for="${id}">${name}</label>
                `;
                return item;
            }

            // --- VALIDATION LOGIC ---

            /**
             * The main validation function, checks the entire form state.
             */
            function validateForm() {
                // Basic field checks
                const isNameFilled = document.getElementById('studentName').value.trim() !== '';
                const isTeacherFilled = document.getElementById('classTeacher').value.trim() !== '';
                const isClassSelected = classSelect.value !== '';
                const isStreamSelected = document.getElementById('stream').value !== '';
                const isTermsChecked = document.getElementById('agreeTerms').checked;

                // Conditional checks
                const isEmailValid = checkEmail();
                const isIndexValid = checkIndexNumber();
                
                // Enable send code button
                sendCodeBtn.disabled = !isEmailValid;

                let canSubmit = false;
                if (!emailVerified) {
                    // Stage 1 validation
                    const isCodeFilled = verificationCodeInput.value.length === 6;
                    canSubmit = isNameFilled && isTeacherFilled && isClassSelected && isStreamSelected && isTermsChecked && isEmailValid && isIndexValid && isCodeFilled;
                } else {
                    // Stage 2 validation
                    const isPasswordValid = checkPassword();
                    const doPasswordsMatch = checkPasswordMatch();
                    const areSubjectsValid = subjectError.style.display === 'none';
                    canSubmit = isNameFilled && isTeacherFilled && isClassSelected && isStreamSelected && isTermsChecked && isEmailValid && isIndexValid && isPasswordValid && doPasswordsMatch && areSubjectsValid;
                }

                // Update submit button state
                submitButton.disabled = !canSubmit;
                submitButton.classList.toggle('active', canSubmit);
            }
            
            function checkEmail() {
                return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(emailInput.value);
            }

            function checkIndexNumber() {
                const isValid = /^[A-Za-z0-9]+$/.test(document.getElementById('indexNumber').value);
                indexNumberError.style.display = isValid || document.getElementById('indexNumber').value.trim() === '' ? 'none' : 'block';
                return isValid;
            }

            function checkPassword() {
                const password = document.getElementById('password').value;
                const isValid = password.length >= 8 && /[A-Z]/.test(password) && /[a-z]/.test(password) && /\d/.test(password) && /[!@#$%^&*()_+]/.test(password);
                passwordError.style.display = isValid || password.trim() === '' ? 'none' : 'block';
                return isValid;
            }

            function checkPasswordMatch() {
                const password = document.getElementById('password').value;
                const confirmPassword = document.getElementById('confirmPassword').value;
                const match = password === confirmPassword;
                confirmPasswordError.style.display = match || confirmPassword.trim() === '' ? 'none' : 'block';
                return match;
            }
            
            /**
             * Validates subject selection count and disables/enables checkboxes.
             */
            function validateSubjectSelection(maxElectives) {
                const electiveContainer = document.getElementById('electiveList');
                const selectedElectives = electiveContainer.querySelectorAll('input[type="checkbox"]:checked');
                const allElectiveCheckboxes = electiveContainer.querySelectorAll('input[type="checkbox"]');
                const counter = document.getElementById('subjectCounter');
                
                const currentCount = selectedElectives.length;
                counter.textContent = `${currentCount}/${maxElectives} Selected`;
                
                const isSelectionComplete = currentCount === maxElectives;
                counter.classList.toggle('complete', isSelectionComplete);
                subjectError.style.display = isSelectionComplete ? 'none' : 'block';
                
                // Disable remaining checkboxes if max is reached
                allElectiveCheckboxes.forEach(cb => {
                    if (!cb.checked) {
                        cb.disabled = isSelectionComplete;
                    }
                });
            }

            // --- UTILITY FUNCTIONS ---
            function showMessage(message, type = 'info') {
                messageDisplay.textContent = message;
                messageDisplay.className = `message-display ${type}`;
                messageDisplay.style.display = 'block';
                setTimeout(() => { messageDisplay.style.display = 'none'; }, 3000);
            }

            function createFloatingBooks() {
                const container = document.getElementById('floatingBooks');
                if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
                const bookIcons = ['book', 'book-open', 'graduation-cap', 'scroll'];
                for (let i = 0; i < 15; i++) {
                    const book = document.createElement('i');
                    book.classList.add('book', 'fas', `fa-${bookIcons[Math.floor(Math.random() * bookIcons.length)]}`);
                    book.style.left = `${Math.random() * 100}%`;
                    book.style.animationDuration = `${Math.random() * 15 + 10}s`;
                    book.style.animationDelay = `${Math.random() * 10}s`;
                    book.style.fontSize = `${Math.random() * 1 + 1}rem`;
                    container.appendChild(book);
                }
            }
        });
    </script>
