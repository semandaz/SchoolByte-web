// --- Download Work File ---
app.post('/api/workfiles/:id/download', authenticateToken, async (req, res) => {
    const { id } = req.params;
    const studentId = req.student.id;
    let bytesDeducted = false;
    let student = null;
    let workFile = null;
    const CLOUD_NAME = "dq5mdy0yq"; // Your hardcoded cloud name

    try {
        // ... (Code to find workFile and student remains the same)
        workFile = await WorkFile.findById(id);
        if (!workFile) { return res.status(404).json({ message: 'File not found' }); }

        student = await Student.findById(studentId);
        if (!student) { return res.status(404).json({ message: 'Student not found' }); }

        if (student.bytes < workFile.costBytes) {
            return res.status(400).json({ 
                message: 'Insufficient bytes',
                required: workFile.costBytes,
                available: student.bytes
            });
        }
        
        // -----------------------------------------------------------
        // 🚀 FIX START: CUSTOM FILENAME + VERSION STRIP 🚀
        // -----------------------------------------------------------

        const fullCloudinaryUrl = workFile.fileUrl; 
        
        // 1. Generate the desired friendly filename (and URL-encode it)
        const customFilename = `${workFile.subject}_${workFile.title.replace(/[^a-z0-9\s-]/gi, '')}.pdf`;
        const encodedFilename = encodeURIComponent(customFilename);

        // 2. Extract the Public ID Path: Strip URL prefix, version number, and file extension
        
        // Get everything after the /upload/ part
        let publicIdPath = fullCloudinaryUrl.split('/upload/')[1]; 
        
        // CRITICAL STEP 1: Remove the version number (e.g., v12345678/) from the path.
        const publicIdWithoutVersion = publicIdPath.replace(/^v\d+\//, ''); 

        // CRITICAL STEP 2: Remove the file extension (.pdf) from the end of the path.
        const publicIdWithoutExtension = publicIdWithoutVersion.replace(/\.pdf$/i, '');

        // 3. FINAL, CORRECT URL ASSEMBLY with custom filename
        // This structure uses the 'fl_attachment:filename' syntax which requires the Public ID 
        // to be completely stripped of the version number and extension.
        const downloadUrl = `https://res.cloudinary.com/${CLOUD_NAME}/raw/upload/fl_attachment:${encodedFilename}/${publicIdWithoutExtension}`;

        console.log(`Download URL constructed (Custom Filename): ${downloadUrl}`);
        
        // -----------------------------------------------------------
        
        // Deduct bytes and save changes (Original logic remains)
        try {
            student.bytes -= workFile.costBytes;
            await student.save();
            bytesDeducted = true;

            workFile.downloadCount += 1;
            await workFile.save();

            console.log(`Bytes deducted: ${workFile.costBytes}. Student now has ${student.bytes} bytes`);
        } catch (saveError) {
            console.error('Error saving student data:', saveError);
            return res.status(500).json({ 
                message: 'Database error during download (byte deduction failed)',
                error: saveError.message 
            });
        }
        
        // -----------------------------------------------------------
        // RETURN URL AND REMOVE STREAMING
        // -----------------------------------------------------------
        
        // Return the URL to the client. The client will handle the redirection/download.
        return res.json({ 
            message: 'File access granted. Redirecting for download.',
            downloadUrl: downloadUrl // This URL contains the custom filename
        });
        
    } catch (error) {
        console.error('Error in download handler:', error);
        
        // Refund bytes if any were deducted (Original logic preserved)
        if (bytesDeducted && student && workFile) {
            try {
                student.bytes += workFile.costBytes;
                await student.save();
                console.log(`Bytes refunded due to handler error: ${workFile.costBytes}`);
            } catch (refundError) {
                console.error('Failed to refund bytes:', refundError);
            }
        }

        if (!res.headersSent) {
            return res.status(500).json({ 
                message: 'Server error during download',
                error: error.message,
                bytesRefunded: bytesDeducted
            });
        }
    }
});
