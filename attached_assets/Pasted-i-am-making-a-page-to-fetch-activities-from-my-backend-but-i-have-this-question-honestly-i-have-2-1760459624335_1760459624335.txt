Of course. What you're describing is a smart and standard way to build websites. You absolutely should use one activities page (activitiesstudentdashboard.html) instead of 21 separate ones.

The solution is to use URL Parameters (also known as query strings) to tell your single activities page which subject's content to load.

Here’s the simple idea: when a student clicks the "Activities" link from the Biology dashboard, you don't just send them to activitiesstudentdashboard.html. Instead, you send them to a URL like this:

activitiesstudentdashboard.html?subject=biology

That little ?subject=biology part at the end is the key. It's like a sticky note you attach to the URL, telling the next page, "Hey, the user is interested in biology!"

How to Implement This: A Step-by-Step Guide
Here is a practical breakdown of how you'd set it up.

## Step 2: Read the Parameter on Your Activities Page
Now, in your activitiesstudentdashboard.html page, you'll need a bit of JavaScript to read that "sticky note" from the URL and then fetch the correct data from your backend.

The modern way to do this is with the URLSearchParams object.

Here is a sample script you would put inside a <script> tag in activitiesstudentdashboard.html:

JavaScript

// This code will run as soon as the activities page loads
document.addEventListener('DOMContentLoaded', () => {

    // 1. Get the parameters from the current page's URL
    const queryString = window.location.search;
    const urlParams = new URLSearchParams(queryString);

    // 2. Get the specific value for the 'subject' parameter
    const subject = urlParams.get('subject'); // This will be 'biology', 'geography', etc.

    // 3. Check if a subject was actually provided in the URL
    if (subject) {
        // You can update the page heading dynamically
        const pageTitle = document.getElementById('activities-heading'); // Assuming you have an <h1 id="activities-heading">
        // Capitalize the first letter for a nice title
        pageTitle.textContent = `${subject.charAt(0).toUpperCase() + subject.slice(1)} Activities`;

        // 4. Call your function to fetch data for that specific subject
        fetchActivities(subject);

    } else {
        // Handle cases where someone navigates to the page directly without a subject
        console.error("No subject was specified in the URL.");
        const contentArea = document.getElementById('activities-container');
        contentArea.innerHTML = '<p>Error: No subject selected. Please go back and select a subject first.</p>';
    }
});

/**
 * Fetches activities from the backend for a given subject.
 * @param {string} subjectName - The name of the subject, e.g., 'biology'.
 */
function fetchActivities(subjectName) {
    // This is where you make the call to your backend API
    // Notice how the 'subjectName' variable is used in the API endpoint URL
    const apiUrl = `https://your-backend.com/api/activities?subject=${subjectName}`;

    console.log(`Fetching data from: ${apiUrl}`); // For debugging

    fetch(apiUrl)
        .then(response => response.json())
        .then(data => {
            // Now you have the activities ONLY for the correct subject
            const contentArea = document.getElementById('activities-container');
            // Clear previous content
            contentArea.innerHTML = '';
            // Loop through the data and display it (this is just an example)
            data.activities.forEach(activity => {
                const activityElement = document.createElement('div');
                activityElement.innerHTML = `<h3>${activity.title}</h3><p>${activity.description}</p>`;
                contentArea.appendChild(activityElement);
            });
        })
        .catch(error => {
            console.error('Error fetching activities:', error);
            const contentArea = document.getElementById('activities-container');
            contentArea.innerHTML = '<p>Sorry, we could not load the activities at this time.</p>';
        });
}

app.post(

    authenticateTeacherToken,
    upload.single('workFilePdf'),
    [
        // Validation for the form data fields
        body('workFileTitle').notEmpty().withMessage('WorkFile title is required.').trim().isLength({ min: 3, max: 200 }),
        body('workFileDescription').optional().isString().withMessage('WorkFile description must be a string.').trim().isLength({ max: 500 }),
        body('workFileSubject').notEmpty().withMessage('WorkFile subject is required.').trim(),
        body('workFileIntendedClass').notEmpty().withMessage('WorkFile intended class is required.').isIn(['S.1', 'S.2', 'S.3', 'S.4', 'S.5', 'S.6']),
        body('workFileCostBytes').notEmpty().isInt({ min: 0 }).withMessage('Cost in Bytes must be a non-negative number.'),
        
        // ✨ KEY FIX: .toBoolean() correctly converts checkbox values like 'on' or 'true' from form data.
        body('applyDownloadWatermark').optional().toBoolean(),

        body('activityJson').notEmpty().withMessage('Activity data (JSON) is required.'),
    ],
    async (req, res) => {
        const errors = validationResult(req);
        if (!errors.isEmpty()) {
            return res.status(400).json({ errors: errors.array() });
        }

        if (!req.file) {
            return res.status(400).json({ message: 'A PDF file for the WorkFile is required.' });
        }

        let activityData;
        try {
            activityData = JSON.parse(req.body.activityJson);
        } catch (e) {
            return res.status(400).json({ message: 'Invalid format for activityJson. It must be a valid JSON string.' });
        }
        
        // Manual, detailed validation for the nested JSON data
        if (!activityData.title || typeof activityData.title !== 'string' || activityData.title.trim().length < 3) {
            return res.status(400).json({ message: 'Activity title is required and must be at least 3 characters.' });
        }
        if (!Array.isArray(activityData.questions) || activityData.questions.length === 0) {

                return res.status(400).json({ message: `Question ${i + 1} must have at least one keyword for marking.` });

        const {
