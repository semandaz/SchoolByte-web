// --- Download Work File ---
app.post('/api/workfiles/:id/download', authenticateToken, async (req, res) => {
    const { id } = req.params;
    const studentId = req.student.id;
    let bytesDeducted = false;
    let student = null;
    let workFile = null;

    try {
        workFile = await WorkFile.findById(id);
        if (!workFile) {
            return res.status(404).json({ message: 'File not found' });
        }

        student = await Student.findById(studentId);
        if (!student) {
            return res.status(404).json({ message: 'Student not found' });
        }

        if (student.bytes < workFile.costBytes) {
            return res.status(400).json({ 
                message: 'Insufficient bytes',
                required: workFile.costBytes,
                available: student.bytes
            });
        }
        
        // -----------------------------------------------------------
        // 🛑 FIX START: CORRECT URL CONSTRUCTION & REMOVE STREAMING 🛑
        // -----------------------------------------------------------

        const fullCloudinaryUrl = workFile.fileUrl; 
        
        // 1. Extract the Public ID *without* the extension (.pdf).
        // This is necessary for the fl_attachment transformation to work correctly.
        const publicIdMatch = fullCloudinaryUrl.match(/\/upload\/(.*)\.pdf/i);

        if (!publicIdMatch || publicIdMatch.length < 2) {
            return res.status(500).json({ message: 'Invalid or unexpected Cloudinary URL structure in database.' });
        }
        
        // publicIdWithoutExtension is the full path (e.g., 'schoolbyte/.../workfile-123')
        const publicIdWithoutExtension = publicIdMatch[1]; 
        
        // 2. Define the desired filename and URL-encode it
        // The title is cleaned and the subject is prefixed for a nice download name.
        const customFilename = `${workFile.subject}_${workFile.title.replace(/[^a-z0-9\s-]/gi, '')}.pdf`;
        const encodedFilename = encodeURIComponent(customFilename);

        // 3. FINAL, CORRECT URL ASSEMBLY (dq5mdy0yq is your Cloud Name)
        // Format: https://res.cloudinary.com/cloudname/raw/upload/fl_attachment:filename/public_id_no_extension
        const downloadUrl = `https://res.cloudinary.com/dq5mdy0yq/raw/upload/fl_attachment:${encodedFilename}/${publicIdWithoutExtension}`;

        console.log(`Download URL constructed: ${downloadUrl}`);
        
        // -----------------------------------------------------------
        // 🛑 FIX END: URL CONSTRUCTION 🛑
        // -----------------------------------------------------------
        
        // Deduct bytes and save changes (Original logic remains)
        try {
            student.bytes -= workFile.costBytes;
            await student.save();
            bytesDeducted = true;

            workFile.downloadCount += 1;
            await workFile.save();

            console.log(`Bytes deducted: ${workFile.costBytes}. Student now has ${student.bytes} bytes`);
        } catch (saveError) {
            console.error('Error saving student data:', saveError);
            return res.status(500).json({ 
                message: 'Database error during download (byte deduction failed)',
                error: saveError.message 
            });
        }
        
        // -----------------------------------------------------------
        // 🛑 FIX: REPLACE ALL STREAMING LOGIC WITH JSON RESPONSE 🛑
        // -----------------------------------------------------------

        // Instead of streaming, we return the URL to the client. 
        // The client will then redirect the browser, which forces the download, 
        // avoiding all server-side streaming errors.
        return res.json({ 
            message: 'File access granted. Redirecting for download.',
            downloadUrl: downloadUrl // Send the working URL back to the client
        });
        
    } catch (error) {
        console.error('Error in download handler:', error);
        
        // Refund bytes if any were deducted (Original logic preserved)
        if (bytesDeducted && student && workFile) {
            try {
                student.bytes += workFile.costBytes;
                await student.save();
                console.log(`Bytes refunded due to handler error: ${workFile.costBytes}`);
            } catch (refundError) {
                console.error('Failed to refund bytes:', refundError);
            }
        }

        if (!res.headersSent) {
            return res.status(500).json({ 
                message: 'Server error during download',
                error: error.message,
                bytesRefunded: bytesDeducted
            });
        }
    }
});
