Snippet 1: The Form Submission Event Listener
This is the core logic that should be updated. It includes console logs for debugging and proper handling of the button's loading state. Replace your existing document.getElementById('uploadQuizForm').addEventListener('submit', ...) block with this updated version.

JavaScript

// Handle form submission
document.getElementById('uploadQuizForm').addEventListener('submit', async function(e) {
    e.preventDefault();
    console.log('Quiz form submission initiated.');

    const statusDiv = document.getElementById('quizStatus');
    const submitBtn = document.getElementById('submitQuizBtn');
    const originalBtnText = submitBtn.textContent;

    submitBtn.textContent = 'Uploading...';
    submitBtn.disabled = true;

    statusDiv.textContent = 'Preparing upload...';
    statusDiv.className = 'mt-4 text-center text-sm font-medium text-gray-600';
    
    const quizQuestions = [];
    const questionBlocks = questionsContainer.querySelectorAll('.question-block');
    questionBlocks.forEach(block => {
        quizQuestions.push({
            questionText: block.querySelector('textarea[name="questionText"]').value,
            subject: quizSubject.value,
            intendedClass: quizClass.value,
            questionType: block.querySelector('select[name="questionType"]').value,
            keywordsForGrading: block.querySelector('input[name="keywords"]').value.split(',').map(k => k.trim()),
            negativeKeywords: block.querySelector('input[name="negativeKeywords"]').value.split(',').map(k => k.trim()).filter(k => k !== '')
        });
    });

    console.log('Questions data collected:', quizQuestions);

    const token = localStorage.getItem('token'); 
    if (!token) {
        console.error('No authentication token found. User is not logged in.');
        statusDiv.textContent = 'Error: Not authenticated. Please log in.';
        statusDiv.className = 'mt-4 text-center text-sm font-medium text-red-600';
        submitBtn.textContent = originalBtnText;
        submitBtn.disabled = false;
        return;
    }
    
    try {
        const response = await fetch('/teacher/quiz-questions', {
            method: 'POST',
            headers: { 
                'Content-Type': 'application/json',
                'Authorization': `Bearer ${token}` 
            },
            body: JSON.stringify({ questions: quizQuestions })
        });

        console.log('Server response received:', response);

        const result = await response.json();
        if (response.ok) {
            statusDiv.textContent = `Successfully uploaded ${result.uploadedCount} questions!`;
            statusDiv.className = 'mt-4 text-center text-sm font-medium text-green-600';
            this.reset();
            questionsContainer.innerHTML = '';
            questionCounter = 0;
            addQuestion();
            for (let i = 0; i < 4; i++) {
                addQuestion();
            }
        } else {
            statusDiv.textContent = `Error: ${result.message}`;
            statusDiv.className = 'mt-4 text-center text-sm font-medium text-red-600';
            console.error('Server responded with an error:', result.message);
        }
    } catch (error) {
        statusDiv.textContent = 'Network error. Please try again.';
        statusDiv.className = 'mt-4 text-center text-sm font-medium text-red-600';
        console.error('Upload Error:', error);
    } finally {
        submitBtn.textContent = originalBtnText;
        submitBtn.disabled = false;
    }
});
Explanation of the changes:
Console Logging (console.log): These lines will print messages to your browser's developer console (press F12 or Ctrl+Shift+I to open it). This is the best way to see where the code stops working.

Loading State: The code now disables the button and changes its text to "Uploading..." while the request is being sent. This provides user feedback and prevents double-clicks.

Authentication Check: It now explicitly checks if a token exists in localStorage. If not, it provides a clear error message. This is a common point of failure if a user is not properly logged in.

Final finally block: The button is now consistently re-enabled and its text reset, whether the upload succeeds or fails.
